  src.settled_date,
  COUNT(src.tx_id) AS source_records,
  COUNT(bi.tx_id) AS bi_records,
  (COUNT(src.tx_id) - COUNT(bi.tx_id)) AS delta
FROM prod_db.settlements src
LEFT JOIN bi_dw.fact_settlements bi 
  ON src.tx_id = bi.tx_id
GROUP BY src.settled_date
HAVING delta != 0;
